// Создаём объект для фильтра Madgwick
Madgwick filter;
// Создаём объект для работы с гироскопом
Gyroscope gyroscope;
// Создаём объект для работы с акселерометром
Accelerometer accelerometer;
// Переменные для данных с гироскопа и акселерометра
float gx, gy, gz, ax, ay, az;
// Переменные для хранения самолётных углов ориентации
float yaw, pitch, roll;
// Переменная для хранения частоты выборок фильтра
float sampleRate = 100;
void setup() {
 // Открываем последовательный порт
 Serial.begin(9600);
//  BluetoothSerial.begin(9600); // открываем порт блютуза
 // Выводим сообщение о начале инициализации
 // Инициализируем гироскоп
 gyroscope.begin();
 // Инициализируем акселерометр
 accelerometer.begin();
 // Инициализируем фильтр
 filter.begin();
 // Выводим сообщение об удачной инициализации
 Serial.println("Initialization completed");
}
void loop() {

 // Запоминаем текущее время
 unsigned long startMillis = millis();
 // Считываем данные с акселерометра в единицах G
 accelerometer.readAccelerationGXYZ(ax, ay, az);
 // Считываем данные с гироскопа в радианах в секунду
 gyroscope.readRotationRadXYZ(gx, gy, gz);
 // Устанавливаем частоту фильтра
 filter.setFrequency(sampleRate);
 // Обновляем входные данные в фильтр
 filter.update(gx, gy, gz, ax, ay, az);
 // Получаем из фильтра углы: yaw, pitch и roll
 yaw = filter.getYawDeg(); // Коорлината Z
 pitch = filter.getPitchDeg(); //  Координата Y
 roll = filter.getRollDeg(); // Координата X
 // Выводим полученные углы Эйлера в Serial-порт
 Serial.print("Z: ");
 Serial.print(yaw);
 Serial.print("\t\t");
 Serial.print("Y: ");
 Serial.print(pitch);
 Serial.print("\t\t");
 Serial.print("X: ");
 Serial.println(roll);
  // BluetoothSerial.print("Z: ");
  // BluetoothSerial.print(yaw);
  // BluetoothSerial.print("\t\t");
  // BluetoothSerial.print("Y: ");
  // BluetoothSerial.print(pitch);
  // BluetoothSerial.print("\t\t");
  // BluetoothSerial.print("X: ");
  // BluetoothSerial.println(roll);
 // Вычисляем затраченное время на обработку данных
 unsigned long deltaMillis = millis() - startMillis;
 // Вычисляем частоту обработки фильтра
 sampleRate = 1000 / deltaMillis;
 
}

#include <nRF24L01.h>
#include <RF24.h>

#include <TroykaIMU.h> // Библиотека для работы с модулями IMU
#include <SoftwareSerial.h>

RF24 radio(9, 10);

Async asyncEngine = Async();

Madgwick filter;
Gyroscope gyroscope; // Создаём объект для работы с гироскопом
Accelerometer accelerometer;// Создаём объект для работы с акселерометром
float gx, gy, gz, ax, ay, az; // Переменные для данных с гироскопа и акселерометра
float yaw, pitch, roll; // Переменные для хранения самолётных углов ориентации
float sampleRate = 100; // Переменная для хранения частоты выборок фильтра

float receivedData[3];

uint8_t pipe;

uint8_t i;
void setup() {

  Serial.begin(115200);
  radio.begin();

  radio.setChannel(112);
  radio.setDataRate(RF24_2MBPS);
  radio.setPALevel(RF24_PA_MIN);
  radio.openReadingPipe(1, 0x0123456789LL);
  radio.startListening();

  gyroscope.begin();
  accelerometer.begin();
  filter.begin();

}
void loop() {

  if (radio.available(&pipe)) {

    radio.read(&receivedData, sizeof(receivedData));

    Serial.print("Данные [ ");
    for (i = 0; i < 3; i++) {
      Serial.print(receivedData[i]);
      Serial.print(' ');
    }
    Serial.print(" ]");
    Serial.println(pipe);
  }
}
